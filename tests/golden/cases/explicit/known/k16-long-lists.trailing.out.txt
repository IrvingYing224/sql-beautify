SELECT
    concat_ws('-', province_name, city_name, district_name, street_name, community_name, building_no, room_no) AS addr
FROM t;
SELECT
    concat_ws(
        '|', coalesce(CAST(user_id AS STRING), ''), coalesce(CAST(order_id AS STRING), ''),
        coalesce(CAST(amount AS STRING), '')
    ) AS line
FROM t;
SELECT
    coalesce(f0, 0) + coalesce(f1, 0) + coalesce(f2, 0) + coalesce(f3, 0) + coalesce(f4, 0) + coalesce(f5, 0) + coalesce(f6, 0) AS total
FROM t;
SELECT
    sum(CASE
            WHEN status IN ('paid', 'shipped', 'delivered', 'completed') THEN
                amount
            ELSE 0
        END) AS paid,
    count(DISTINCT user_id, order_id, product_id, sku_id, warehouse_id, region_id, channel_id) AS n
FROM t;
