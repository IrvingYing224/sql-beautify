select user_id,
  row_number() over (partition by user_id order by event_time desc) as rn,
  sum(amount) over (partition by user_id order by event_time rows between unbounded preceding and current row) as running,
  lag(amount, 1, 0) over (partition by user_id order by event_time) as prev_amt,
  avg(amount) over (partition by user_id order by event_time range between 3 preceding and 3 following) as av
from orders
